\begin{document}

    scale only axis,
    axis equal image,
    axis lines=middle,
    grid=major,
    grid style={black!10},

    xlabel={$x$},

  ]
    \nextgroupplot[
      title={Funció irracional},

      ytick={-1,0,1,2,3,4,5},

      \node[anchor=north east, font=\scriptsize, fill=white, inner sep=0.5pt]

        at (axis cs:0.25,2.65) {$g(x)=e^{-x}$};

      title={Funció logarítmica},
